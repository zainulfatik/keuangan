function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ss.getSheets();
    var allTransactions = [];
    
    for (var i = 0; i < sheets.length; i++) {
      var sheet = sheets[i];
      var name = sheet.getName();
      if (name.match(/^\d{4}$/)) {
        var data = sheet.getDataRange().getValues();
        for (var r = 0; r < data.length; r++) {
          var row = data[r];
          if (row[0] && typeof row[0] === 'string' && row[0].match(/^\d{4}-\d{2}-\d{2}$/)) {
            allTransactions.push({
              id: 'tx_sheet_' + r + '_' + name,
              date: row[0],
              type: row[1],
              wallet: row[2],
              category: row[3],
              amount: Number(row[4]) || 0,
              notes: row[5] || ''
            });
          }
        }
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({ 
      status: 'success', 
      transactions: allTransactions 
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      status: 'error', 
      message: err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  
  try {
    var data = JSON.parse(e.postData.contents);
    var dateObj = new Date(data.date || new Date());
    var year = dateObj.getFullYear().toString();
    
    var monthNames = ["Januari", "Februari", "Maret", "April", "Mei", "Juni", "Juli", "Agustus", "September", "Oktober", "November", "Desember"];
    var monthName = monthNames[dateObj.getMonth()];
    var fullMonthTitle = "REKAP TRANSAKSI BULAN " + monthName.toUpperCase() + " " + year;
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(year);
    
    if (!sheet) {
      sheet = ss.insertSheet(year);
      createMonthHeader(sheet, fullMonthTitle);
    }
    
    var lastRow = sheet.getLastRow();
    if (lastRow > 0) {
      var needsNewMonthHeader = checkIsNewMonth(sheet, dateObj);
      if (needsNewMonthHeader) {
        sheet.appendRow([""]);
        createMonthHeader(sheet, fullMonthTitle);
      }
    } else {
      createMonthHeader(sheet, fullMonthTitle);
    }
    
    var formattedAmount = Number(data.amount) || 0;
    var rowData = [
      data.date,
      data.type,
      data.wallet,
      data.category,
      formattedAmount,
      data.notes || ''
    ];
    
    sheet.appendRow(rowData);
    var currentRow = sheet.getLastRow();
    formatTransactionRow(sheet, currentRow, data.type);
    
    updateSummarySheet(ss);

    return ContentService.createTextOutput(JSON.stringify({ 
      status: 'success', 
      message: 'Transaksi berhasil disimpan' 
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ 
      status: 'error', 
      message: err.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function updateSummarySheet(ss) {
  var summarySheet = ss.getSheetByName("REKAP SALDO");
  if (!summarySheet) {
    summarySheet = ss.insertSheet("REKAP SALDO");
  }
  summarySheet.clear();
  
  summarySheet.appendRow(["REKAP SALDO PER DOMPET"]);
  summarySheet.getRange("A1:D1").merge().setBackground("#0F172A").setFontColor("#FFFFFF").setFontWeight("bold").setHorizontalAlignment("center");
  
  summarySheet.appendRow(["Nama Dompet", "Total Pemasukan", "Total Pengeluaran", "Saldo Saat Ini"]);
  summarySheet.getRange("A2:D2").setBackground("#4F46E5").setFontColor("#FFFFFF").setFontWeight("bold").setHorizontalAlignment("center");
  
  var defaultWallets = ["Tas", "Seabank", "Jok", "Lemari", "BNI", "BRI"];
  var walletMap = {};
  defaultWallets.forEach(function(w){ walletMap[w] = { inc: 0, exp: 0 }; });
  
  var sheets = ss.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    var sh = sheets[i];
    if (sh.getName().match(/^\d{4}$/)) {
      var data = sh.getDataRange().getValues();
      for (var r = 0; r < data.length; r++) {
        var row = data[r];
        if (row[0] && typeof row[0] === 'string' && row[0].match(/^\d{4}-\d{2}-\d{2}$/)) {
          var type = row[1];
          var wallet = row[2];
          var amt = Number(row[4]) || 0;
          
          if (type === 'Pemasukan') {
            if (!walletMap[wallet]) walletMap[wallet] = { inc: 0, exp: 0 };
            walletMap[wallet].inc += amt;
          } else if (type === 'Pengeluaran') {
            if (!walletMap[wallet]) walletMap[wallet] = { inc: 0, exp: 0 };
            walletMap[wallet].exp += amt;
          } else if (type === 'Transfer') {
            var parts = wallet.split('➔');
            if (parts.length === 2) {
              var wFrom = parts[0].trim();
              var wTo = parts[1].trim();
              if (!walletMap[wFrom]) walletMap[wFrom] = { inc: 0, exp: 0 };
              if (!walletMap[wTo]) walletMap[wTo] = { inc: 0, exp: 0 };
              walletMap[wFrom].exp += amt;
              walletMap[wTo].inc += amt;
            }
          }
        }
      }
    }
  }
  
  var rowIdx = 3;
  for (var wName in walletMap) {
    var inc = walletMap[wName].inc;
    var exp = walletMap[wName].exp;
    var bal = inc - exp;
    summarySheet.appendRow([wName, inc, exp, bal]);
    summarySheet.getRange(rowIdx, 2, 1, 3).setNumberFormat('"Rp "#,##0');
    rowIdx++;
  }
  
  summarySheet.setColumnWidth(1, 150);
  summarySheet.setColumnWidth(2, 160);
  summarySheet.setColumnWidth(3, 160);
  summarySheet.setColumnWidth(4, 180);
}

function createMonthHeader(sheet, monthTitle) {
  sheet.appendRow([monthTitle]);
  var titleRow = sheet.getLastRow();
  var titleRange = sheet.getRange(titleRow, 1, 1, 6);
  titleRange.merge();
  titleRange.setBackground("#1E293B");
  titleRange.setFontColor("#FFFFFF");
  titleRange.setFontWeight("bold");
  titleRange.setHorizontalAlignment("center");
  
  var headers = ["Tanggal", "Tipe", "Dompet", "Kategori", "Jumlah (Rp)", "Keterangan"];
  sheet.appendRow(headers);
  var headerRow = sheet.getLastRow();
  var headerRange = sheet.getRange(headerRow, 1, 1, 6);
  headerRange.setBackground("#4F46E5");
  headerRange.setFontColor("#FFFFFF");
  headerRange.setFontWeight("bold");
  headerRange.setHorizontalAlignment("center");
  
  sheet.setColumnWidth(1, 110);
  sheet.setColumnWidth(2, 110);
  sheet.setColumnWidth(3, 150);
  sheet.setColumnWidth(4, 160);
  sheet.setColumnWidth(5, 140);
  sheet.setColumnWidth(6, 250);
}

function checkIsNewMonth(sheet, currentDateObj) {
  var lastRow = sheet.getLastRow();
  for (var r = lastRow; r >= 1; r--) {
    var val = sheet.getRange(r, 1).getValue();
    if (val && val !== "" && typeof val === 'string' && val.match(/^\d{4}-\d{2}-\d{2}$/)) {
      var prevDate = new Date(val);
      if (prevDate.getMonth() !== currentDateObj.getMonth() || prevDate.getFullYear() !== currentDateObj.getFullYear()) {
        return true;
      }
      return false;
    }
  }
  return false;
}

function formatTransactionRow(sheet, row, type) {
  var range = sheet.getRange(row, 1, 1, 6);
  if (type === 'Pemasukan') {
    range.setBackground("#E6F4EA");
    range.setFontColor("#137333");
  } else if (type === 'Pengeluaran') {
    range.setBackground("#FCE8E6");
    range.setFontColor("#C5221F");
  } else {
    range.setBackground("#E8F0FE");
    range.setFontColor("#1A73E8");
  }
  sheet.getRange(row, 5).setNumberFormat('"Rp "#,##0');
}
